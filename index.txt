    <script>
        // Dữ liệu mẫu dự phòng nếu chưa có file data.js
        if (typeof window.appData === 'undefined') {
            window.appData = [

                { sap: "100119196", description: "Push-in L-connector QS-8H-6", partNumber: "153042", location: "080801", quantity: 9, unit: "PC", price: "15.00", typeOrder: "Standard", reorderPoint: "2", minLotSize: "5" },
                { sap: "100119383", description: "connector Y QSY-6-4", partNumber: "QSY-6-4", location: "041305", quantity: 5, unit: "PC", price: "8.20", typeOrder: "Batch", reorderPoint: "10", minLotSize: "20" },
                { sap: "100120072", description: "Manual tensioner VERTIKALSPAN...", partNumber: "202-U", location: "051002", quantity: 0, unit: "PC", price: "150.00", typeOrder: "Special", reorderPoint: "1", minLotSize: "1" }
            ];
        }

        let currentFilteredData = [...window.appData];
        let selectedIndex = -1;

        function renderTable(data) {
            const tbody = document.getElementById('tableBody');
            tbody.innerHTML = '';
            
            data.forEach((item, index) => {
                const tr = document.createElement('tr');
                if (index === selectedIndex) tr.className = 'selected';
                tr.innerHTML = `
                    <td style="width: 100px;">${item.sap || ''}</td>
                    <td>${item.description || ''}</td>
                    <td style="width: 120px;">${item.partNumber || ''}</td>
                    <td style="width: 80px;">${item.location || ''}</td>
                    <td style="width: 70px; text-align: right;">${item.quantity ?? ''}</td>
                    <td style="width: 50px;">${item.unit || 'PC'}</td>
                `;
                tr.onclick = () => selectRow(index, data);
                tbody.appendChild(tr);
            });

            document.getElementById('recordsFound').innerText = data.length.toLocaleString('de-DE');
            document.getElementById('totalRecords').innerText = window.appData.length.toLocaleString('de-DE');
        }

        function selectRow(index, data) {
            selectedIndex = index;
            renderTable(data);
            
            const item = data[index];
            if (item) {
                document.getElementById('detSap').value = item.sap || '';
                document.getElementById('detLocation').value = item.location || '';
                document.getElementById('detStock').value = item.quantity ?? '';
                document.getElementById('detPrice').value = item.price || '';
                document.getElementById('detTypeOrder').value = item.typeOrder || '';
                document.getElementById('detReorderPoint').value = item.reorderPoint || '';
                document.getElementById('detMinLotSize').value = item.minLotSize || '';
            }
        }

        function handleSearch() {
            const keyword = document.getElementById('searchInput').value.toLowerCase().trim();
            const searchField = document.querySelector('input[name="searchField"]:checked').value;
            const qtyZeroOnly = document.getElementById('chkQtyZero').checked;

            currentFilteredData = window.appData.filter(item => {
                // Lọc theo Quantity > 0
                if (qtyZeroOnly && Number(item.quantity) <= 0) return false;

                // Lọc theo từ khóa ô nhập
                if (!keyword) return true;

                let targetValue = "";
                if (searchField === 'sap') targetValue = item.sap;
                else if (searchField === 'description') targetValue = item.description;
                else if (searchField === 'partNo') targetValue = item.partNumber;
                else if (searchField === 'bmNo') targetValue = item.bmNo;

                return String(targetValue || '').toLowerCase().includes(keyword);
            });

            selectedIndex = currentFilteredData.length > 0 ? 0 : -1;
            renderTable(currentFilteredData);
            if (selectedIndex >= 0) selectRow(selectedIndex, currentFilteredData);
        }

        function handleClear() {
            document.getElementById('searchInput').value = '';
            document.getElementById('chkQtyZero').checked = false;
            document.getElementById('chkMultiSearch').checked = false;
            currentFilteredData = [...window.appData];
            selectedIndex = currentFilteredData.length > 0 ? 0 : -1;
            renderTable(currentFilteredData);
            if (selectedIndex >= 0) selectRow(selectedIndex, currentFilteredData);
        }

        function prevImage() {
            alert('Chức năng chuyển ảnh trước.');
        }

        function nextImage() {
            alert('Chức năng chuyển ảnh tiếp theo.');
        }

        function exportData() {
            let csvContent = "data:text/csv;charset=utf-8,SAP,Description,Part Number,Location,Quantity,Unit\n";
            currentFilteredData.forEach(row => {
                csvContent += `"${row.sap}","${row.description}","${row.partNumber}","${row.location}","${row.quantity}","${row.unit}"\n`;
            });
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "exported_materials.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        // Khởi chạy lần đầu khi mở trang
        window.onload = () => {
            if (window.appData && window.appData.length > 0) {
                currentFilteredData = [...window.appData];
                selectRow(0, currentFilteredData);
            }
        };
    </script>

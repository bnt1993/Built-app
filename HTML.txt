  <script>
    function escapeHTML(value) {
      return String(value ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
    }

    // ================= CẤU HÌNH THƯ MỤC Ổ MẠNG =================
    const duongDanOMang = "data.js";
    const THU_MUC_HINH = "HinhAnh/"; // Thư mục chứa ảnh trên ổ mạng
    const SO_ANH_TOI_DA = 10;        // Tự động quét từ _01 đến _10
    // ==========================================================

    let tatCaDuLieu = [];
    let tieuDeCot = [];
    let dongDangChonIndex = -1;
    let danhSachAnhDangXem = [];
    let viTriAnhDangXem = 0;
    let mocCapNhatCu = "";

    function taiDuLieu() {
      const scriptCu = document.getElementById("script-data");
      if (scriptCu) scriptCu.remove();

      const s = document.createElement("script");
      s.id = "script-data";
      s.src = duongDanOMang + "?t=" + Date.now();

      s.onload = function() {
        if (window.thoiGianCapNhat !== mocCapNhatCu) {
          mocCapNhatCu = window.thoiGianCapNhat;
          tatCaDuLieu = window.duLieuCongTy || [];
          if (tatCaDuLieu.length > 0) {
            tieuDeCot = tatCaDuLieu[0];
            veTieuDe();
            locDuLieu();
          }
          document.getElementById("trang-thai-cap-nhat").textContent = "Cập nhật lúc: " + mocCapNhatCu;
        }
      };
      document.body.appendChild(s);
    }

    function veTieuDe() {
      document.getElementById("dau-bang").innerHTML = "<tr>" + tieuDeCot.map(c => `<th>${escapeHTML(c)}</th>`).join("") + "</tr>";
    }

    function locDuLieu() {
      if (tatCaDuLieu.length < 2) return;
      const tuKhoa = document.getElementById("o-tim-kiem").value.trim().toLowerCase();
      const cheDoTim = document.querySelector('input[name="truongTim"]:checked').value;
      const chiTonKho = document.getElementById("chk-ton-kho").checked;

      let html = "";
      let soBanGhiTimThay = 0;

      for (let i = 1; i < tatCaDuLieu.length; i++) {
        const dong = tatCaDuLieu[i];
        let khop = true;

        // Lọc theo từ khóa và trường tìm kiếm
        if (tuKhoa) {
          if (cheDoTim === "all") {
            khop = dong.some(cell => String(cell).toLowerCase().includes(tuKhoa));
          } else {
            // Xác định vị trí cột tương ứng (Ví dụ giả định: Cột 0 là Sap, Cột 1 là Description, Cột 2 là Part No...)
            let colIndex = cheDoTim === "sap" ? 0 : (cheDoTim === "desc" ? 1 : 2);
            let val = String(dong[colIndex] || "").toLowerCase();
            khop = val.includes(tuKhoa);
          }
        }

        // Lọc số lượng > 0 (giả định cột Stock/Quantity nằm ở vị trí thứ 3 hoặc 4, điều chỉnh theo file Excel của bạn)
        if (khop && chiTonKho) {
          let tonKhoVal = parseFloat(dong[3]) || 0; // Thay đổi số 3 nếu cột tồn kho khác vị trí
          if (tonKhoVal <= 0) khop = false;
        }

        if (khop) {
          soBanGhiTimThay++;
          let selectedClass = (i === dongDangChonIndex) ? "selected" : "";
          html += `<tr class="${selectedClass}" onclick="chonDong(${i}, this)">` +
                  dong.map(cell => `<td>${escapeHTML(cell)}</td>`).join("") +
                  `</tr>`;
        }
      }

      document.getElementById("than-bang").innerHTML = html;
      document.getElementById("trang-thai-ban-ghi").textContent = `Records Found: ${soBanGhiTimThay} / ${tatCaDuLieu.length - 1}`;
    }

    function chonDong(index, element) {
      document.querySelectorAll("#than-bang tr").forEach(tr => tr.classList.remove("selected"));
      element.classList.add("selected");
      dongDangChonIndex = index;

      const dongData = tatCaDuLieu[index];
      const maSap = (dongData[0] || "").trim(); // Cột đầu tiên là Mã SAP

      // Hiển thị thông tin chi tiết sang panel bên phải
      let htmlChiTiet = "";
      for (let c = 0; c < tieuDeCot.length; c++) {
        htmlChiTiet += `<div class="detail-item"><label>${escapeHTML(tieuDeCot[c])}</label><span>${escapeHTML(dongData[c] ?? '')}</span></div>`;
      }
      document.getElementById("luoi-chi-tiet").innerHTML = htmlChiTiet;

      // Tải hình ảnh tương ứng với Mã SAP
      taiBoSuuTapAnh(maSap);

      document.getElementById("thong-tin-meta").innerHTML = `User: Hệ thống nội bộ<br>Last update: ${escapeHTML(mocCapNhatCu)}`;
    }

    // Tải bộ sưu tập ảnh _01, _02...
    function taiBoSuuTapAnh(maSap) {
      danhSachAnhDangXem = [];
      viTriAnhDangXem = 0;
      capNhatGiaoDienAnh();

      if (!maSap) return;

      // Thử dò từ _01 đến _10
      for (let k = 1; k <= SO_ANH_TOI_DA; k++) {
        const hauTo = k < 10 ? "0" + k : "" + k;
        kiemTraVaThemAnh(`${maSap}_${hauTo}.jpg`, `${maSap}_${hauTo}.png`);
      }

      // Kiểm tra thêm ảnh gốc không có hậu tố
      kiemTraVaThemAnh(`${maSap}.jpg`, `${maSap}.png`);
    }

    function kiemTraVaThemAnh(tenJpg, tenPng) {
      const imgTest = new Image();
      const urlJpg = THU_MUC_HINH + encodeURIComponent(tenJpg);

      imgTest.onload = function() {
        if (!danhSachAnhDangXem.includes(urlJpg)) {
          danhSachAnhDangXem.push(urlJpg);
          danhSachAnhDangXem.sort();
          capNhatGiaoDienAnh();
        }
      };
      imgTest.onerror = function() {
        const imgPng = new Image();
        const urlPng = THU_MUC_HINH + encodeURIComponent(tenPng);
        imgPng.onload = function() {
          if (!danhSachAnhDangXem.includes(urlPng)) {
            danhSachAnhDangXem.push(urlPng);
            danhSachAnhDangXem.sort();
            capNhatGiaoDienAnh();
          }
        };
        imgPng.src = urlPng;
      };
      imgTest.src = urlJpg;
    }

    function capNhatGiaoDienAnh() {
      const imgEl = document.getElementById("anh-chinh");
      const btnPrev = document.getElementById("btn-anh-truoc");
      const btnNext = document.getElementById("btn-anh-sau");
      const counterEl = document.getElementById("dem-anh");
      const tong = danhSachAnhHienTai = danhSachAnhDangXem.length;

      if (tong > 0) {
        if (viTriAnhDangXem >= tong) viTriAnhHienTai = 0;
        imgEl.src = danhSachAnhDangXem[viTriAnhHienTai];
        counterEl.textContent = `${viTriAnhHienTai + 1} / ${tong} images`;

        const hienNut = tong > 1 ? "block" : "none";
        btnPrev.style.display = hienNut;
        btnNext.style.display = hienNut;
      } else {
        imgEl.src = "";
        counterEl.textContent = "0/0 images";
        btnPrev.style.display = "none";
        btnNext.style.display = "none";
      }
    }

    function doiAnhChiTiet(buoc, e) {
      e.stopPropagation();
      const tong = danhSachAnhDangXem.length;
      if (tong <= 1) return;
      viTriAnhHienTai = (viTriAnhHienTai + buoc + tong) % tong;
      capNhatGiaoDienAnh();
    }

    function phongToAnh() {
      const src = document.getElementById("anh-chinh").src;
      if (src) {
        document.getElementById("lightbox-img").src = src;
        document.getElementById("lightbox").style.display = "flex";
      }
    }

    function lamMoi() {
      document.getElementById("o-tim-kiem").value = "";
      document.querySelector('input[name="truongTim"][value="all"]').checked = true;
      document.getElementById("chk-ton-kho").checked = false;
      locDuLieu();
    }

    function xuatExcel() {
      alert("Tính năng xuất dữ liệu đã sẵn sàng trên hệ thống mạng!");
    }

    // Tự động nạp dữ liệu và làm mới mỗi 5 giây
    taiDuLieu();
    setInterval(taiDuLieu, 5000);
  </script>
